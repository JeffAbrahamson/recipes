\dish{拍黄瓜}
\hiragana{pāi huángguā}
\altdish{Smashed cucumber salad}
\serves{4}
\prep{20 minutes + 50 minutes rest}
\source{openai-chatgpt}

\begin{ingredients}
  \ingrS{2}{}{cucumbers, medium (Persian or mini ideal), seeded if large French}
  \ingr{\fracH}{tsp}{salt}
  \ingr{1}{\tbsp}{light soy sauce}
  \ingr{1}{\tbsp}{Chinkiang black vinegar}
  \ingr{1--2}{tsp}{\hyperref[laziyou]{辣椒油} (chili oil)}
  \ingr{\fracH}{tsp}{sesame oil}
  \ingr{\fracH}{tsp}{sugar}
  \ingrS{2--3}{cloves}{garlic, grated or finely minced}
  \ingr{}{}{toasted sesame seeds (optional)}
  \ingr{}{}{cilantro, chopped (optional)}
\end{ingredients}


\begin{recipe}
  \begin{enumerate}

  \item Smash the cucumbers with the flat side of a cleaver (or a
    rolling pin) until they split into irregular chunks with cracks.
    Cut into 3--4~cm pieces.

  \item Toss with the salt, let sit 15--20~minutes. Drain and squeeze
    out excess liquid. Chill 20--30~minutes.

  \item Combine remaining ingredients except garnish (sesame seeds,
    cilantro); taste and adjust.

  \item Toss cucumbers with the dressing just before serving. Rest
    10--15~minutes. Garnish with sesame seeds and cilantro.

  \end{enumerate}
\end{recipe}

\notes{} For serving alongside \hyperref[tian-shui-mian]{甜水面}, halve the sugar
(\fracQ{} tsp) so the dish contrasts with the sweeter noodle sauce
rather than competing with it.
