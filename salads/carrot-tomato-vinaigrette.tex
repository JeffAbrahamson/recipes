\dish{Carrot and Tomato Salad}
\label{carrot-tomato-vinaigrette}
\altdish{Grated carrot and tomato salad}
\altdish{Tomato and carrot salad}
\altdish{Mustard, grated carrot and tomato salad}
\serves{2}
\prep{15 min}
\sourcep{chatgpt}

\begin{ingredients}
  \ingr{1}{}{tomato, cut into small wedges}
  \ingr{}{}{salt}
  \ingrS{2}{}{carrots, grated}
  \ingr{1--1\fracH}{Tbsp}{wine vinegar or lemon juice}
  \ingr{2}{Tbsp}{olive oil}
  \ingr{}{}{Dijon mustard (optional)}
  \ingr{}{}{black pepper}
  \ingrS{}{drops}{toasted sesame oil}
  \ingr{}{}{parsley, coriander or chives, chopped (optional)}
\end{ingredients}


\begin{recipe}
  \begin{enumerate}

  \item Salt the tomato wedges lightly.  Let sit 5--10~min so they
    release juice.

  \item Whisk the tomato juice with the vinegar, a little mustard if
    using, the olive oil, and a few drops of sesame oil.

  \item Toss the dressing with the grated carrots.  Fold in the
    tomato wedges and herbs.  Season with black pepper.

  \end{enumerate}
\end{recipe}

\notes{} Keep the dressing noticeably tart --- both carrot and
tomato are sweet.  Pairs with \hyperref[pasta-poivron-doubanjiang]{Red
Pepper \& Doubanjiang Pasta}, whose rich, umami sauce wants a fresh,
acidic contrast; the sesame oil echoes its fermented-chile character
without repeating it.
