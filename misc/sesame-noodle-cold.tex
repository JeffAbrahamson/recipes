\dish{Japanese Cold Sesame Noodles}
\altdish{Cold sesame noodles, Japanese}
\altdish{Sesame noodles, cold, Japanese}
\altdish{Noodles, cold sesame, Japanese}
\altdish{冷やしごまだれ麺}
\hiragana{ひやしごまだれめん}
\serves{2}
\prep{30 minutes}
\source{openai-chatgpt}

\begin{ingredients}
  \ingrS{200}{g}{fresh ramen noodles (or 100--120 g dry)}
  \ingr{45}{ml}{toasted sesame paste (neri goma)}
  \ingr{30}{ml}{soy sauce}
  \ingr{15}{ml}{rice vinegar}
  \ingr{15}{g}{sugar}
  \ingr{10}{ml}{sesame oil}
  \ingr{30--45}{ml}{\hyperref[dashi2]{dashi}}
  \ingr{1}{\tsp}{grated ginger}
  \ingr{1}{small clove}{garlic, grated (optional)}
  \ingrS{1--2}{\tsp}{rayu (optional)}
  \ingr{1}{}{cucumber, julienned}
  \ingr{}{}{spring onions, sliced}
  \ingr{}{}{toasted white sesame seeds}
  \ingr{}{}{nori, cut into strips}
  \ingr{1}{}{cooked chicken breast, shredded (optional)}
  \ingr{2}{}{eggs (optional)}
  \ingrS{}{}{tomato wedges (optional)}
\end{ingredients}


\begin{recipe}
  \begin{enumerate}

  \item Whisk the sesame paste, soy sauce, vinegar, sugar, sesame oil,
    dashi, ginger, garlic, and rayu until smooth.  Refrigerate for
    20--30~minutes.

  \item If using the eggs, make a thin omelette and cut into strips.

  \item Cook the noodles.  Rinse under cold water, immerse briefly in
    ice water, and drain.

  \item Toss with two-thirds of the sauce.  Add the toppings and spoon
    over the remaining sauce.

  \end{enumerate}
\end{recipe}
