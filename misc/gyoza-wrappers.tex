\dish{餃子の皮}
\hiragana{ぎょうざのかわ}
\altdish{Gyoza wrappers}
\altdish{Gyoza skins}
\makes{30}
\prep{30~min + rest (30--60~min)}
\source{openai-chatgpt}

\begin{ingredients}
  \ingr{200}{g}{flour (T55)}
  \ingr{\fracH}{\tsp}{salt}
  \ingrS{100}{ml}{boiling water}
  \ingr{}{}{cornstarch for dusting}
\end{ingredients}


\begin{recipe}
  \begin{enumerate}

  \item Mix flour and salt.  Stir in the water with chopsticks.

  \item When cool enough to handle, knead until smooth, firm, and not
    sticky (5--8~minutes).

  \item Cover tightly and rest 30--60~minutes.

  \item Roll into a log and cut into 30 pieces.  Flatten each.

  \item Roll to 8--9~cm circles, about 1~mm thick in the centre and
    thinner at the edges.

  \item Dust with cornstarch and stack.

  \end{enumerate}
\end{recipe}

\notes Keeps 24~hours in the fridge, tightly wrapped.
