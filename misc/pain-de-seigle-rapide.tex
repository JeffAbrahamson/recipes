\dish{Pain de seigle rapide}
\altdish{Seigle, pain de (rapide)}
\altdish{Bread, rye (quick)}
\altdish{Rye bread, quick}
\altdish{Levain, pain de seigle au}
%\serves{}
\makes{900 g}
\prep{levain la veille + 5 heures}
\source{jma}

\stage{Levain solide}

\begin{ingredients}
  \ingr{60}{g}{levain liquide}
  \ingr{80}{ml}{eau}
  \ingr{120}{g}{farine de seigle}
\end{ingredients}

\begin{recipe}
  \begin{enumerate}

  \item La veille, mélanger levain, eau et farine.  Couvrir et laisser
    reposer la nuit à température ambiante.

  \end{enumerate}
\end{recipe}

\stage{Pâte}

\begin{ingredients}
  \ingr{320}{ml}{eau}
  \ingr{}{pincée}{levure sèche}
  \ingr{40}{g}{gluten}
  \ingr{400}{g}{farine de seigle}
  \ingr{10}{g}{sel}
\end{ingredients}

\begin{recipe}
  \begin{enumerate}

  \item Le lendemain, délayer le levain solide dans l'eau et la
    levure.  Ajouter gluten, farine et sel.  Pétrir.

  \item Laisser reposer 2$\,$\fracH{} heures, avec pli après 1 et
    2~heures.

  \item Pré-former.  Laisser reposer 15--20 minutes.

  \item Former en bâtard (ou boule).  Laisser reposer 1\fracH--2
    heures.

  \item Préchauffer le four à 240\degreeC.

  \item Grigner.  Cuire à 240\degreeC{} puis 180\degreeC{} pendant
    30--35 minutes.

  \item Laisser refroidir sur une volette au moins 10~minutes avant de
    couper.

  \end{enumerate}
\end{recipe}
