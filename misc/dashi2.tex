\dish{一番だしと二番だし}
\alsodish{Ichiban and Niban Dashi}
\altdish{いちばんだしとにばんだし}
\makes{500 ml each}
\prep{45 minutes}
\source{openai-chatgpt}
\label{dashi2}

\stage{一番だし}

\begin{ingredients}
  \ingr{500}{ml}{water}
  \ingr{5--8}{g}{kombu}
  \ingr{10--15}{g}{katsuobushi}
\end{ingredients}


\begin{recipe}
  \begin{enumerate}

  \item Soak the kombu in the water for 20--30~minutes.

  \item Heat slowly.  Remove the kombu at 80--90\textdegree C, before
    the water boils.

  \item Bring to a boil and turn off the heat.  Add the katsuobushi
    and steep for 2--3~minutes (5~minutes for thick shavings).

  \item Strain without pressing the flakes.  Reserve the kombu and
    katsuobushi for the niban dashi.

  \end{enumerate}
\end{recipe}


\stage{二番だし}

\begin{ingredients}
  \ingr{500--750}{ml}{water}
  \ingr{}{reserved}{kombu and katsuobushi}
  \ingr{}{}{katsuobushi (optional)}
\end{ingredients}


\begin{recipe}
  \begin{enumerate}

  \item Bring the water, reserved kombu, and katsuobushi slowly to a
    gentle simmer.  Simmer for 5--10~minutes.

  \item Add a small handful of fresh katsuobushi during the final
    minute, if using.  Strain.

  \end{enumerate}
\end{recipe}
