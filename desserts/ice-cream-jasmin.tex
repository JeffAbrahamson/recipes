\dish{Glace au thé jasmin}
\altdish{Jasmin ice cream}
\altdish{Ice cream, jasmin}
%\serves{}
\fait{1 litre}
\prep{20 minutes plus congélation}
\source{jma}

\stage{Base}

\begin{ingredients}
  \ingr{30}{g}{thé jasmin}
  \ingr{230}{ml}{eau}
  \ingrS{450}{g}{crème}
  \ingr{150}{g}{sugar}
  \ingrS{1,5--2,0}{g}{sel}
  \ingrS{3}{}{jaune d'oeufs}
  \ingrS{2}{\tbsp}{vodka}
  \ingr{60}{g}{meringue italienne~(page \pageref{label:meringue-italienne-gelato})}
\end{ingredients}


\begin{recipe}
  \begin{enumerate}

  \item Infuser le jasmin dans la crème et l'eau au frigo pendant
    8--12 heures.

  \item Passer au tamise.  Chauffer la crème, sucre, et sel.  Ajouter
    lentement aux oeufs en fouettant.  Remettre sur feu doux est faire
    une crème anglaise (80\degreeC).

  \item Ajouter le vodka et turbiner, ajoutant la meringue au moment
    que la préparation commence à épaissir.

  \end{enumerate}
\end{recipe}
