\dish{Sorbet au chocolat}
\altdish{Chocolat, sorbet}
\altdish{Gelato, chocolat}
%\serves{}
\fait{400 ml}
\prep{20 minutes + temps de macération + congélation}
\source{jma}

\begin{ingredients}
  \ingr{320}{ml}{eau}
  \ingr{50}{g}{sucre semoule}
  \ingr{40}{g}{sucre inverti}
  \ingrS{20}{g}{cacao poudre}
  \ingrS{110--120}{g}{chocolat (originelle : Valrhona Madagascar)}
  \ingrS{1}{\tbsp}{grand marnier}
  \ingrS{60}{g}{meringue italienne~(page \pageref{label:meringue-italienne-gelato})}
\end{ingredients}

\begin{recipe}
  \begin{enumerate}

  \item Mélanger eau, sucre et cacao.  Faire frémir 2--5~minutes.
    Mixer.

  \item Faire fondre le chocolat, ajouter mélange cacao pour faire une
    ganache.  Mixer.

  \item Laisser refroidir 24~heures au réfrigérateur.

  \item Turbiner, ajoutant le meringue mi-congélation.

  \end{enumerate}
\end{recipe}
