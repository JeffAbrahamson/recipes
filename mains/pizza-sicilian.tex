\dish{Sicilian Pizza Dough}
\altdish{Pizza dough, Sicilian}
\serves{8}
%\makes{}
\prep{20 minutes + 7 hours}
\source{modernist.bread}

\begin{ingredients}
  \textbf{Poolish}\\
  \ingr{50}{g}{water}
  \ingr{50}{g}{bread flour}
  \ingrS{0.05}{g}{dry yeast}
  \textbf{Dough}\\
  \ingr{330}{g}{water}
  \ingrS{3}{g}{dry yeast}
  \ingrS{535}{g}{bread flour}
  \ingr{12}{g}{salt}
  \ingrS{5}{g}{olive oil}
  \ingr{}{}{olive oil for pan}
\end{ingredients}

\begin{recipe}
  \begin{enumerate}

  \item \textbf{Poolish:} Combine flour and water with a pinch of
    yeast (first three ingredients), stir well and ferment 12~hours.

  \item \textbf{Dough:} Combine water and yeast.  Add poolish.  Add
    flour.  Mix to shaggy mass.  Allow to autolyse, 30~minutes.

  \item Add salt and olive oil.  Mix to medium gluten development.
    Cover and allow to ferment 2\fracH~hours, with 1~book fold after
    each hour.  Check for full gluten development.  If the dough
    gasses too quickly, put it in the refrigerator for bulk fermentation.

  \item Drizzle a baking pan copiously with oliver oil.  The oil will
    fry the dough whilst baking, which is key for the characteristic
    crispy taste.  Spread the dough in the pan by stippling with
    fingers.  Allow to proof 1\fracH--2~hours (27\degreeC) or up to
    3~hours (21\degreeC), stippling once again mid-proof.

  \item Bake at 220\degreeC{} for about 20~minutes.  Cool for several
    minutes in the pain so that the oil re-absorbs and the crust
    becomes crisp, then transfer to a wire rack.  Serve at room
    temperature or while still very slightly warm.

  \end{enumerate}
\end{recipe}

\stage{Toppings}

\begin{itemize}
\item Very thin smear of tomato paste diluted with olive oil, with a
  touch of chili flakes.
\item Red lentil (about a half cup dry + 1 cup water to cook), on the
  dry side of cooked.
\item Onion (700~g / 4~onions) cooked in olive oil with a pinch of
  salt until nicely golden.
\item Feta (crumbled, on the light side)
\item Semi-dried tomato (chopped, not too heavy)
\item \textbf{(after baking)} Red onion marinated in red wine vinegar, lightly
  rinsed, pre-dressed with olive oil.
\item 10--15 pistachios, crushed irregularly, maybe lightly toasted.
\item A couple mint leaves (very sparingly), torn.
\end{itemize}
