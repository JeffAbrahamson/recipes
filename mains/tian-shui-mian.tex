\dish{甜水面}
\hiragana{tián shuǐ miàn}
\serves{2}
\prep{30 minutes (+ dough rest if made from scratch)}
\source{openai-chatgpt}
\label{tian-shui-mian}

\begin{ingredients}
  \ingrS{250--300}{g}{fresh thick wheat noodles (4--5~mm), or dough below}
  \ingr{2}{\tbsp}{light soy sauce}
  \ingr{2}{\tbsp}{Chinese dark brown sugar (or 1\fracH{} \tbsp{} molasses + 1 \tbsp{} sugar)}
  \ingr{2}{\tbsp}{chili oil (\hyperref[laziyou]{辣椒油}), with sediment}
  \ingr{1}{tsp}{sesame paste}
  \ingr{2--3}{cloves}{garlic, finely grated}
  \ingr{\fracH}{tsp}{ground roasted Sichuan pepper}
  \ingr{\fracH}{tsp}{sesame oil}
  \ingrS{1--3}{\tbsp}{noodle cooking water}
  \ingr{\fracH}{tsp}{black vinegar (optional)}
  \ingrS{}{}{pinch MSG (optional)}
  \ingr{}{}{scallions, chopped (optional)}
  \ingr{}{}{toasted sesame seeds, to finish}
\end{ingredients}


\begin{recipe}
  \begin{enumerate}

  \item If making the dough: mix 300~g bread flour, 150~g water, 3~g
    salt.  Knead 8--10~minutes until firm. Rest 30~minutes, knead
    briefly, rest another 30~minutes. Roll into ropes finger-thick,
    cut into 20--25~cm lengths.  The noodles will expand somewhat in
    water.

  \item Boil the noodles until chewy but cooked through (fresh: about
    5--7~minutes). Reserve a few tablespoons of cooking water.

  \item Whisk together the the remaining ingredients until smooth.
    Thin with cooking water until it coats a spoon (but still quite
    thick) as needed.

  \item Drain noodles, toss vigorously with the sauce until every
    strand is coated. Top with sesame seeds and scallions. Serve
    immediately.

  \end{enumerate}
\end{recipe}

\notes{} Don't substitute doubanjiang --- its fermented flavor
overwhelms the clean sweet-garlic profile. For newcomers to Sichuan
sweetness, cut the sugar 20--30\%.
