\dish{Crispy Tofu, Caramelised Onion \& Mustard Pasta}
\altdish{Tofu, onion, and mustard pasta}
\altdish{Pasta with crispy tofu, mustard, and onions}
\altdish{Mustard crème fraîche sauce, over crispy tofu pasta}
\serves{2}
\prep{30 min}
\sourcep{chatgpt}

\begin{ingredients}
  \ingr{200}{g}{firm tofu}
  \ingr{1}{Tbsp}{soy sauce}
  \ingr{1}{Tbsp}{cornflour}
  \ingr{}{}{neutral or olive oil}
  \ingrS{160--170}{g}{pasta}
  \ingr{2}{}{onions, thinly sliced}
  \ingr{}{}{salt}
  \ingr{2}{cloves}{garlic, minced}
  \ingr{1}{tsp}{Dijon mustard}
  \ingr{2--3}{Tbsp}{crème fraîche}
  \ingr{}{}{black pepper (more than seems reasonable)}
  \ingr{\fracH}{tsp}{soy sauce}
  \ingrS{\fracH--1}{tsp}{lemon juice or vinegar}
  \ingr{}{}{smoked paprika or chilli flakes (optional)}
\end{ingredients}


\begin{recipe}
  \begin{enumerate}

  \item Pat tofu dry, cut into 2~cm cubes.  Toss with 1~Tbsp soy
    sauce, then with the cornflour.

  \item Fry in a generous film of oil over medium-high heat, turning
    occasionally, until deeply golden on several sides, 7--10~min.
    Remove and reserve.

  \item Cook pasta in salted water.  Reserve a mugful of pasta water
    before draining.

  \item In the tofu pan, fry the onions in a little more oil with a
    pinch of salt, 6--8~min, until soft and browning.  Add garlic for
    the last minute.

  \item Turn heat down.  Add mustard, crème fraîche, black
    pepper, and about 60~ml pasta water.  Stir into a loose, glossy
    sauce.  Add \fracH~tsp soy sauce and the lemon juice or vinegar.
    Add paprika or chilli flakes if using.

  \item Add drained pasta and toss, adding pasta water until the
    sauce coats it rather than sitting thick.  Fold in two-thirds of
    the tofu; top with the rest so some stays crisp.

  \item Taste before salting: pasta water, mustard, and soy may
    already be enough.

  \end{enumerate}
\end{recipe}
