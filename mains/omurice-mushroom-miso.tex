\dish{オムライス}
\altdish{Omurice, miso-butter mushroom}
\serves{2}
\prep{30 minutes}
\source{anthropic-claude}
\label{omurice-mushroom-miso}

\stage{Rice}

\begin{ingredients}
  \ingr{200}{g}{mushrooms, mixed, sliced}
  \ingr{1}{\tbsp}{neutral oil}
  \ingr{\fracH}{}{onion, finely diced}
  \ingr{1}{small}{carrot, finely diced}
  \ingr{1}{clove}{garlic, finely chopped}
  \ingrS{350--400}{g}{cold cooked Japanese rice}
  \ingr{1}{\tbsp}{red or mixed miso}
  \ingr{1}{\tbsp}{soy sauce}
  \ingr{1}{\tbsp}{sake or dry white wine}
  \ingrS{1--2}{tsp}{ketchup}
  \ingrS{10}{g}{butter}
  \ingr{\fracH}{tsp}{toasted sesame oil}
  \ingr{1}{}{spring onion, sliced}
  \ingr{}{}{black pepper}
\end{ingredients}

\begin{recipe}
  \begin{enumerate}

  \item Brown mushrooms in oil in a hot pan, uncrowded.  Set aside
    \fracQ{} for topping.

  \item Lower heat.  Cook onion and carrot until the onion is
    translucent.  Add garlic (30~seconds).

  \item Add rice, break apart, fry 2~minutes.

  \item Mix miso, soy sauce, sake and ketchup.  Push rice aside, pour
    mixture onto the pan, caramelise 10--15~seconds, then stir in.

  \item Add butter, sesame oil, spring onion and plenty of pepper.
    Taste before salting.  Keep warm.

  \end{enumerate}
\end{recipe}

\stage{Sauce}

\begin{ingredients}
  \ingr{2}{\tbsp}{ketchup}
  \ingr{1}{\tbsp}{Worcestershire sauce}
  \ingr{1}{\tbsp}{sake, wine or water}
  \ingr{\fracH}{tsp}{soy sauce}
  \ingr{}{}{small knob of butter}
\end{ingredients}

\begin{recipe}
  \begin{enumerate}

  \item In the rice pan, cook ketchup and Worcestershire 30~seconds.
    Add sake, simmer briefly.  Off heat, whisk in butter and soy sauce.

  \end{enumerate}
\end{recipe}

\stage{Omelettes}

\begin{ingredients}
  \ingr{4--6}{}{eggs (2--3 per person)}
  \ingr{1}{\tbsp}{milk or cream}
  \ingr{}{}{salt}
  \ingr{}{}{butter}
  \ingr{}{}{shichimi t\=ogarashi (optional)}
\end{ingredients}

\begin{recipe}
  \begin{enumerate}

  \item For each person, beat 2--3 eggs lightly with milk and salt.

  \item Melt butter in a non-stick pan over medium-high heat.  Add
    eggs, stir fast with chopsticks while shaking the pan.  Stop when
    curds are very soft and the surface still wet.

  \item Mound rice on a plate.  Either let the bottom of the omelette
    set, slide it over the rice and tuck the edges under; or, tampopo
    style, shape into a soft oval with a barely set centre, set on the
    rice and slit lengthwise to open.

  \item Spoon sauce around, top with reserved mushrooms and shichimi.

  \end{enumerate}
\end{recipe}
