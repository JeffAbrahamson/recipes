\dish{おにぎり}
\altdish{焼きおにぎり}
\altdish{Onigiri}
\altdish{Yaki onigiri}
\prep{1~h (+ 15--30 minutes for yaki)}
\source{jma}
\label{onigiri}

\stage{Rice}

\begin{ingredients}
  \ingr{3}{cups}{Japanese rice, uncooked}
  \ingr{}{}{salt}
  \ingr{}{}{filling (below)}
\end{ingredients}

\begin{recipe}
  \begin{enumerate}

  \item Cook the rice, lightly salted.

  \item \emph{Yaki:} mix in the seasoning below.  \emph{Plain:} no
    seasoning.

  \item Form onigiri around \fracH--1~tsp filling.  Plain onigiri: stop
    here.

  \end{enumerate}
\end{recipe}

\stage{Filling}

\begin{ingredients}
  \ingr{2}{\tbsp}{sesame paste (nerigoma, or Chinese zhīmajiàng)}
  \ingr{1}{\tbsp}{miso}
  \ingr{1}{tsp}{mirin}
  \ingr{\fracH--1}{tsp}{ginger, finely grated}
\end{ingredients}

Thick paste, not a sauce.

\stage{焼きおにぎり}

\begin{ingredients}
  \ingr{3}{\tbsp}{soy sauce}
  \ingr{1}{\tbsp}{sesame oil}
  \ingr{1}{\tbsp}{mirin (optional)}
  \ingrS{\fracQ}{tsp}{salt}
  \ingr{}{}{oil, light}
  \ingr{}{}{soy sauce, miso, mirin, equal parts, for glaze}
\end{ingredients}

\begin{recipe}
  \begin{enumerate}

  \item Seasoning (soy sauce, sesame oil, mirin, salt) goes into the
    rice before forming.

  \item Let formed onigiri firm up and dry 15--30~minutes.

  \item Fry in a lightly oiled pan over medium heat until crusted,
    3--5~minutes per side.

  \item Brush thinly with glaze.  Grill briefly at lower heat.

  \end{enumerate}
\end{recipe}
