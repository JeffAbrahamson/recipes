\dish{なすと焼豚のトマトラーメン}
\hiragana{なすとやきぶたのトマトラーメン}
\altdish{Tomato ramen with eggplant}
\altdish{Ramen, tomato (eggplant, chāshū)}
\serves{2}
\prep{30~min}
\source{nas-tomato-ramen}


\stage{Soup}

\begin{ingredients}
  \ingr{600}{ml}{dashi}
  \ingr{200}{g}{canned chopped tomatoes}
  \ingr{\fracQ}{}{red onion, thinly sliced}
  \ingrS{1}{}{dried chilli, seeded}
  \ingr{2}{\tbsp}{soy sauce}
  \ingr{2}{\tbsp}{sake}
  \ingr{1}{\tbsp}{mirin}
  \ingr{1}{\tbsp}{sesame oil}
  \ingr{\fracH}{\tsp}{salt}
  \ingr{1}{\tsp}{garlic, grated}
  \ingr{1}{\tsp}{ginger, grated}
  \ingr{\fracQ}{\tsp}{coarse black pepper}
\end{ingredients}

\begin{recipe}
  \begin{enumerate}

  \item Bring everything to the boil. Keep hot over low heat.

  \end{enumerate}
\end{recipe}


\stage{Toppings and assembly}

\begin{ingredients}
  \ingr{3}{}{eggplants}
  \ingrS{}{}{neutral oil for frying}
  \ingr{100--120}{g}{\hyperref[chashu]{chāshū}, sliced 3--5~mm}
  \ingrS{1}{}{tomato, sliced 5~mm}
  \ingr{220}{g}{thin \hyperref[ラーメン]{ramen noodles}}
  \ingr{}{}{lemon, in quarter slices}
  \ingr{}{}{broccoli sprouts}
  \ingr{}{}{grated parmesan}
\end{ingredients}

\begin{recipe}
  \begin{enumerate}

  \item Halve the eggplants lengthwise; score the flesh in a
    crosshatch. Shallow-fry in 1~cm of oil at 180\degreeC, both
    sides, until tender.

  \item Cook and drain the noodles. Divide among bowls with the
    soup. Arrange tomato, chāshū and lemon; top with eggplant and
    sprouts. Dust with parmesan.

  \end{enumerate}
\end{recipe}

\notes Hot oil keeps the eggplant from soaking it up.
